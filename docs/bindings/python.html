Python Binding

pip install cx-lang. ctypes under the hood. Pythonic API on top.

Install

shellpip install cx-lang
# Or from source:
git clone https://github.com/cx-home/cx
cd cx/lang/python
pip install -e .

Parse and Emit

pythonimport cx

tree = cx.parse(open('pizza.cx').read())
print(cx.emit(tree, to='json'))
print(cx.emit(tree, to='yaml'))
print(cx.canonical(open('pizza.cx').read()))

Tree Access

pythontree = cx.parse('[shop name=Pepe [pizza name=Margherita price=12]]')

# Pythonic: attributes look like dict; children look like a list
shop = tree[0]
print(shop['name'])              # 'Pepe'
print(shop[0]['name'])           # 'Margherita'
print(shop[0]['price'])          # 12 (an int, not a string)

# Iterate children
for pizza in tree.find('pizza'):
    print(pizza['name'], pizza['price'])

Run a CXL Template

pythonimport cx

program = open('receipt.cxl').read()
data    = open('order.cx').read()
print(cx.eval(program, data=data))

# With include resolution
print(cx.eval(program, data=data, include_root='/proj/templates'))

Format Conversion

pythonyaml_text = open('config.yaml').read()
cx_text   = cx.from_yaml(yaml_text)

# Or one-shot
out = cx.convert(yaml_text, from_='yaml', to='cx')

Schema Validation

pythondiagnostics = cx.validate(
    open('order.cx').read(),
    schema=open('order.cxs').read(),
    mode='strict',
)
for d in diagnostics:
    print(d.severity, d.message, d.path)

Streaming

pythonwith open('events.cx', 'rb') as f:
    for event in cx.stream(f):
        if event.kind == 'element_start' and event.name == 'error':
            print(event.attrs)

with cx.stream_writer(open('out.cx', 'wb')) as w:
    for record in fetch_records():
        w.write(record)

Type Mapping

CX type to Python type: int to int, bigint to int, decimal to decimal.Decimal, float / f32 / f64 to float, bool to bool, string to str, date to datetime.date, datetime to datetime.datetime, bytes to bytes, null to None.
